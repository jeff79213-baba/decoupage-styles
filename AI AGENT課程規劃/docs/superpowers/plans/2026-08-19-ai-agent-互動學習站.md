**Goal:** 建立單檔 HTML 課堂互動學習站，把 9 章 AI Agent 教材改寫為互動模組，含章節選擇器、三種主題、全互動（視覺化+模擬+問答）。

**Architecture:** `index.html` 承載全部 CSS/HTML/UI 邏輯（經典 `<script>`，file:// 可直接開啟），純函式抽到 `app-logic.js`（UMD 格式，同時可被瀏覽器與 Vitest 載入），測試用 Vitest。先建骨架+主題系統（Task 2），再逐章加入模組 HTML（Task 3-11），最後做章節選擇器與課堂導覽（Task 12）。

**Tech Stack:** 純 HTML5 + CSS3 + Vanilla JS（無框架）、Vitest（單元測試）、Inter Google Fonts、CSS custom properties 做主題切換。

## Global Constraints

- **單檔體驗**：`index.html` 雙擊即可開啟（file://）。CSS 全部內嵌 `<style>`，UI 邏輯內嵌 `<script>`，唯一外部檔案是 `app-logic.js`（必須與 index.html 同目錄）。
- **純函式必須可測**：所有無 DOM 依賴的邏輯（章節篩選、分頁計算、進度計算、主題變數、計分）都放 `app-logic.js`，用 UMD 格式（`window.AILearn` + `module.exports`），Vitest 可 import。
- **localStorage keys（專案前綴規則）**：`ailearn-selected`（章節索引陣列）、`ailearn-theme`（green|dark|fresh）、`ailearn-progress`（結業完成陣列）。
- **三主題 CSS 變數**（`data-theme` 切換）：green（主色 #76B947, 底 #F1F5F2, Inter）、dark（底 #0F172A/#1E293B, 螢光藍 #38BDF8 + 紫 #A78BFA, 淺色文字）、fresh（底 #E0F2FE, 主色 #0284C7, 柔和投影）。
- **9 章模組**：每章一個 `.module` div，含至少 1 種視覺化 + 1 個模擬操作 + 3 題問答 + 「本節重點」概念卡。比喻/版本與既有教材一致（Node=小型工廠、npm=應用程式商店、opencode=AI助理本體、Git=時光機+保險箱、gh=遙控器、Firebase=上架機器、17 skill=14+2+1）。
- **問答格式**：`.quiz-q` 設 `data-answer`（正確選項 0-based 索引），所有選項按鈕 `onclick="checkQuiz(this)"`；`checkQuiz` 由 app-logic 的邏輯支援（見 Task 1 介面）。
- **章節選擇器**：設定面板 9 個 checkbox + 全選/全不選 + 「進入課堂」；未選章節在課堂模式完全不可見。
- **課堂導覽**：底部固定列「◀ 上一章 / 下一章 ▶」+「第 N/M 章」+ 章節下拉選單；鍵盤 ← → 翻頁。
- **防 XSS**：動態內容一律 `textContent`，禁用未淨化 `innerHTML`。
- 內容語言：繁體中文，目標讀者為完全無程式基礎者。
- 檔案路徑：`AI AGENT課程互動站\`（在 firebase雲端資料夾根目錄下）。

```js
(function (root, factory) {
  if (typeof module === 'object' && module.exports) {

  }

  var THEMES = {
    green: {
      '--primary': '#76B947',
      '--primary-light': '#e8f5e0',
      '--primary-dark': '#5a9a32',
      '--bg': '#F1F5F2',

    },
    dark: {
      '--primary': '#38BDF8',

      '--primary-dark': '#7DD3FC',
      '--bg': '#0F172A',
      '--dark': '#1E293B',
      '--card': '#1E293B',
      '--text': '#E2E8F0',
      '--text-secondary': '#94A3B8',
      '--border': '#334155',
      '--border-light': '#334155'
    },
    fresh: {
      '--primary': '#0284C7',
      '--primary-light': '#E0F2FE',
      '--primary-dark': '#075985',
      '--bg': '#F0F9FF',
      '--dark': '#0C4A6E',
      '--card': '#FFFFFF',
      '--text': '#0F172A',
      '--text-secondary': '#64748B',
      '--border': '#BAE6FD',
      '--border-light': '#E0F2FE'
    }
  };

  function filterSelected(allIds, selectedIds) {
    return allIds.filter(function (id) { return selectedIds.indexOf(id) !== -1; });
  }

  function pageNext(current, selected) {
    var i = selected.indexOf(current);
    if (i === -1 || i === selected.length - 1) return current;
    return selected[i + 1];
  }

  function pagePrev(current, selected) {
    var i = selected.indexOf(current);
    if (i <= 0) return current;
    return selected[i - 1];
  }

  function progressPct(selectedIds, completedIds) {
    if (selectedIds.length === 0) return 0;
    var done = selectedIds.filter(function (id) { return completedIds.indexOf(id) !== -1; }).length;
    return Math.round((done / selectedIds.length) * 100);
  }

  function scoreResult(correct, total) {
    var pct = total === 0 ? 0 : Math.round((correct / total) * 100);
    var label = pct === 100 ? '滿分！' : (pct >= 60 ? '及格' : '再練習');
    return { pct: pct, label: label };
  }

  function themeVars(name) {
    return THEMES[name] || THEMES.green;
  }

  return {
    THEMES: THEMES,
    themeVars: themeVars,
    filterSelected: filterSelected,
    pageNext: pageNext,
    pagePrev: pagePrev,
    progressPct: progressPct,
    scoreResult: scoreResult
  };
});
```

```js
function setTheme(name) {
  var vars = AILearn.themeVars(name);
  var root = document.documentElement;
  for (var k in vars) { root.style.setProperty(k, vars[k]); }
  document.body.setAttribute('data-theme', name);
  try { localStorage.setItem('ailearn-theme', name); } catch (e) {}
}
function initTheme() {
  var saved = 'green';
  try { saved = localStorage.getItem('ailearn-theme') || 'green'; } catch (e) {}
  setTheme(saved);
  var btns = document.querySelectorAll('.theme-switch button');
  for (var i = 0; i < btns.length; i++) {
    btns[i].classList.toggle('active', btns[i].getAttribute('data-theme') === saved);
  }
}
```

```js
function toggleTree(el) {
```

```js
function checkQuiz(btn) {
  var q = btn.closest('.quiz-q');
  var opts = q.querySelectorAll('.quiz-opt');
  var correct = parseInt(q.getAttribute('data-answer'), 10);
  for (var i = 0; i < opts.length; i++) opts[i].disabled = true;
  for (var i = 0; i < opts.length; i++) {
    if (i === correct) opts[i].classList.add('correct');
  }
  if (btn !== opts[correct]) btn.classList.add('wrong');
  var explain = q.querySelector('.quiz-explain');
  if (explain) explain.classList.add('show');
  var module = btn.closest('.module');
  if (module) updateModuleScore(module);
}
function updateModuleScore(module) {
  var qs = module.querySelectorAll('.quiz-q');
  var correct = 0;
  for (var i = 0; i < qs.length; i++) {
    var opts = qs[i].querySelectorAll('.quiz-opt');
    var ci = parseInt(qs[i].getAttribute('data-answer'), 10);
    if (opts[ci] && opts[ci].classList.contains('correct')) correct++;
  }
  var scoreEl = module.querySelector('.quiz-score');
  if (!scoreEl) return;
  scoreEl.classList.add('show');
  var r = AILearn.scoreResult(correct, qs.length);
  scoreEl.textContent = '得分：' + correct + '/' + qs.length + '（' + r.pct + '%）— ' + r.label;
}
```

```js
var slideState = { list: [], index: 0 };

function enterClassroom() {
  var boxes = document.querySelectorAll('#settingsView .chapter-option input');
  var sel = [];
  for (var i = 0; i < boxes.length; i++) {
    if (boxes[i].checked) sel.push(parseInt(boxes[i].value, 10));
  }
  if (sel.length === 0) { alert('請至少選擇一章'); return; }
  try { localStorage.setItem('ailearn-selected', JSON.stringify(sel)); } catch (e) {}
  slideState.list = AILearn.filterSelected([0,1,2,3,4,5,6,7,8], sel);
  slideState.index = 0;
  showSlide(slideState.list[slideState.index]);
  document.getElementById('settingsView').style.display = 'none';
  document.getElementById('slideView').style.display = 'block';
}
function goNext() { slideState.index = slideState.list.indexOf(AILearn.pageNext(slideState.list[slideState.index], slideState.list)); showSlide(slideState.list[slideState.index]); }
function goPrev() { slideState.index = slideState.list.indexOf(AILearn.pagePrev(slideState.list[slideState.index], slideState.list)); showSlide(slideState.list[slideState.index]); }
function showSlide(id) {
  var mods = document.querySelectorAll('#slideView .module');
  for (var i = 0; i < mods.length; i++) mods[i].classList.toggle('active', mods[i].id === 'mod' + id);
  var m = slideState.list.length;
  var pos = slideState.list.indexOf(id) + 1;
  document.getElementById('slideIndicator').textContent = '第 ' + pos + '/' + m + ' 章';
  var sel = document.getElementById('chapterSelect');
  sel.innerHTML = '';
  slideState.list.forEach(function (cid) {
    var opt = document.createElement('option');
    opt.value = cid;
    opt.textContent = '第 ' + cid + ' 章';
    if (cid === id) opt.selected = true;
    sel.appendChild(opt);
  });
  var pct = Math.round((pos / m) * 100);
  document.getElementById('progressFill').style.width = pct + '%';
  window.scrollTo(0, 0);
}
function onChapterSelect() {
  var cid = parseInt(document.getElementById('chapterSelect').value, 10);
  slideState.index = slideState.list.indexOf(cid);
  showSlide(cid);
}
function backToSettings() {
  document.getElementById('slideView').style.display = 'none';
  document.getElementById('settingsView').style.display = 'block';
}
```
